    **5. Update class list rendering to pass new props and handle Ctrl+click:**

    ```tsx
    classes.map(cls => (
      <SymbolTreeItem
        key={cls.id}
        symbol={cls}
        isSelected={selectedClass?.id === cls.id}
        isPinned={pinnedClassIds.has(cls.id)}
        isFocused={focusedClassId === cls.id}
        onClick={(e: React.MouseEvent) => {
          if (e.ctrlKey || e.metaKey) {
            // Ctrl+click: toggle pin without changing focus
            if (pinnedClassIds.has(cls.id)) {
              unpinClass(cls.id)
            } else {
              pinClass(cls.id)
            }
          } else {
            // Normal click: select (which also pins via store's selectClass)
            selectClass(cls.id)
          }
        }}
        onTogglePin={(e: React.MouseEvent) => {
          e.stopPropagation() // don't trigger row click
          if (pinnedClassIds.has(cls.id)) {
            unpinClass(cls.id)
          } else {
            pinClass(cls.id)
          }
        }}
      />
    ))
    ```
