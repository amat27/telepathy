    **5. Rewrite the `GraphView` component:**

    ```typescript
    export function GraphView() {
      const {
        multiGraph,
        pinnedClassIds,
        pinnedClasses,
        focusedClassId,
        selectedMember,
        selectedMembers,
        isLoadingGraph,
        isLoadingPin,
        focusClass,
        selectClass,
        pinClass,
      } = useAppStore()

      const { nodes: layoutedNodes, edges: layoutedEdges } = useMemo(
        () => buildMultiClassFlowElements(
          multiGraph,
          pinnedClassIds,
          pinnedClasses,
          focusedClassId,
          selectedMember?.id ?? null,
          selectedMembers,
        ),
        [multiGraph, pinnedClassIds, pinnedClasses, focusedClassId, selectedMember, selectedMembers]
      )

      // Use controlled mode (pass nodes/edges directly, no useNodesState/useEdgesState)
      // Only track user drag changes via onNodesChange
      const [nodes, setNodes, onNodesChange] = useNodesState(layoutedNodes)
      const [edges, setEdges, onEdgesChange] = useEdgesState(layoutedEdges)

      // Sync when layout changes
      useMemo(() => {
        setNodes(layoutedNodes)
        setEdges(layoutedEdges)
      }, [layoutedNodes, layoutedEdges, setNodes, setEdges])

      const { fitView } = useReactFlow()

      // Fit view when nodes change (with slight delay for render)
      useEffect(() => {
        if (layoutedNodes.length > 0) {
          setTimeout(() => fitView({ padding: 0.2, duration: 300 }), 50)
        }
      }, [layoutedNodes, fitView])

      const onNodeClick = useCallback(
        (_: React.MouseEvent, node: Node) => {
          if (pinnedClassIds.has(node.id)) {
            // Clicking a pinned node → focus it
            focusClass(node.id)
          } else {
            // Clicking a non-pinned node → navigate to it (which also pins it)
            selectClass(node.id)
          }
        },
        [pinnedClassIds, focusClass, selectClass]
      )

      if (isLoadingGraph || isLoadingPin) {
        return <div className="graph-placeholder">Loading graph...</div>
      }

      if (pinnedClassIds.size === 0) {
        return (
          <div className="graph-placeholder">
            <p>Select a class to view its hierarchy</p>
          </div>
        )
      }

      return (
        <div className="graph-view">
          <ReactFlow
            nodes={nodes}
            edges={edges}
            onNodesChange={onNodesChange}
            onEdgesChange={onEdgesChange}
            onNodeClick={onNodeClick}
            nodeTypes={nodeTypes}
            fitView
            fitViewOptions={{ padding: 0.3 }}
            minZoom={0.1}
            maxZoom={2}
            proOptions={{ hideAttribution: true }}
          >
            <Background color="var(--border-color)" gap={20} />
            <Controls
              showInteractive={false}
              style={{ background: 'var(--bg-surface)', borderColor: 'var(--border-color)' }}
            />
            <MiniMap
              nodeColor={(n) => {
                if (n.id === focusedClassId) return 'var(--text-accent)'
                if (pinnedClassIds.has(n.id)) return 'var(--color-class)'
                return 'var(--text-muted)'
              }}
              style={{ background: 'var(--bg-secondary)' }}
            />
          </ReactFlow>
        </div>
      )
    }
    ```
